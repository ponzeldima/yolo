================================================================================
  ПІДКЛЮЧЕННЯ PPM ЧЕРЕЗ AUDIO JACK (3.5mm) — Radiomaster TX12 + macOS
================================================================================

Дата: квітень 2026
Модель пульта: Radiomaster TX12 (EdgeTX)
Метод: PPM-сигнал через аудіовихід Mac → тренерський порт пульта


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1. ЩО ПОТРІБНО
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - Mac з аудіовиходом 3.5mm (або USB аудіо-адаптер)
  - Кабель 3.5mm TRS male ↔ 3.5mm TRS male (звичайний AUX-кабель)
  - Radiomaster TX12 з прошивкою EdgeTX

  Опціонально (якщо Mac не має 3.5mm):
  - USB-C → 3.5mm аудіо-адаптер (Apple або сторонній)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  2. РОЗПІНОВКА 3.5mm TRS РОЗ'ЄМУ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TRS (Tip-Ring-Sleeve) — стандартний стерео-штекер:

        ┌─────┐
    Tip │ PPM │ ← Сигнал PPM (аудіо Left channel)
        ├─────┤
   Ring │  —  │ ← не використовується (аудіо Right channel)
        ├─────┤
 Sleeve │ GND │ ← Земля (спільна)
        └─────┘

  На Radiomaster TX12 тренерський порт (3.5mm jack на задній панелі):
    - Tip  = PPM сигнал IN/OUT
    - Ring = не підключений (NC) або живлення (на деяких пультах)
    - Sleeve = GND

  ⚠️  УВАГА: На деяких пультах Ring може мати напругу (5V–8V для живлення
     зовнішнього модуля). У TX12 він зазвичай NC, але НЕ ЗАМИКАЙ Ring на GND
     і не підключай до нього нічого навмисно.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  3. СХЕМА ПІДКЛЮЧЕННЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Mac (Headphone Out)                    Radiomaster TX12 (Trainer Port)
  ┌──────────────────┐                   ┌──────────────────┐
  │                  │    AUX Cable      │                  │
  │  3.5mm Out  ─────┼───────────────────┼─── 3.5mm In      │
  │  (Tip = Left Ch) │  TRS ↔ TRS       │  (Tip = PPM In)  │
  │  (Sleeve = GND)  │                   │  (Sleeve = GND)  │
  └──────────────────┘                   └──────────────────┘

  Це просто AUX-кабель. Нічого паяти не потрібно.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  4. НАЛАШТУВАННЯ macOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. System Settings → Sound → Output:
     - Обери "Headphone" або свій USB аудіо-адаптер
     - Гучність: 70–100% (PPM потребує достатнього рівня сигналу)

  2. Переконайся що інші звуки (сповіщення, музика) вимкнені або паузовані —
     будь-який зайвий аудіо-сигнал зіпсує PPM-кадри!

  3. Запусти тест:
     python rc_test_channels.py --method ppm

  4. Якщо потрібно обрати конкретний аудіопристрій:
     python rc_test_channels.py --method ppm --audio-device 2
     (список пристроїв виводиться при запуску)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  5. НАЛАШТУВАННЯ EdgeTX НА RADIOMASTER TX12
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Крок 1: Увімкнути Trainer Mode
  ─────────────────────────────
    - Довга кнопка SYS (System) → "Trainer"
    - Mode: "Master / Jack"  (пульт = майстер, отримує PPM через jack)

  Крок 2: Налаштувати маппінг каналів
  ─────────────────────────────────────
    У розділі Trainer налаштуй для кожного каналу:

    CH1 (Roll/Aileron)   → Source: CH1  → Mode: "Replace" або "+=" (Add)
    CH2 (Pitch/Elevator) → Source: CH2  → Mode: "Replace" або "+="
    CH3 (Throttle)       → Source: CH3  → Mode: "Replace" або "+="
    CH4 (Yaw/Rudder)     → Source: CH4  → Mode: "Replace" або "+="

    "Replace" — повністю замінити палички на вхідні дані
    "+=" (Add) — додати до існуючих палички (мікшування)

    Для тесту рекомендую "Replace" — так відразу побачиш рух у Channel Monitor.

  Крок 3: Перевірити
  ──────────────────
    - Перейди в Model → Channel Monitor (довга кнопка MDL)
    - Запусти скрипт на Mac
    - Бари каналів CH1–CH4 мають рухатися синусоїдально


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  6. ДІАГНОСТИКА ПРОБЛЕМ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Проблема: Канали не рухаються
  ──────────────────────────────
  - Перевір гучність Mac (має бути 70–100%)
  - Перевір що кабель — TRS (стерео), а не TRRS (з мікрофоном)
  - Перевір Trainer Mode = "Master / Jack"
  - Спробуй змінити полярність у rc_ppm_audio.py:
      PPM_POLARITY = -1.0  (замість 1.0)

  Проблема: Канали хаотично скачуть
  ──────────────────────────────────
  - Зменш гучність до 50–70%
  - Переконайся що немає інших звуків (вимкни сповіщення)
  - Спробуй USB аудіо-адаптер (вбудований DAC може мати шум)

  Проблема: Канали зміщені (центр не 1500)
  ─────────────────────────────────────────
  - У Trainer → Calibration зроби калібрування тренерського входу
  - Або підлаштуй PULSE_WIDTH_US у rc_ppm_audio.py


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  7. БЕЗПЕКА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ✅ БЕЗПЕЧНО:
  - AUX-кабель не може пошкодити ні Mac, ні пульт
  - Аудіо-вихід Mac — це низьковольтний сигнал (~1V peak)
  - Тренерський порт TX12 розрахований на прийом PPM

  ⚠️ НЕ РОБИ:
  - Не підключай зовнішнє живлення до Ring контакту
  - Не використовуй кабелі з мікрофонним роз'ємом (TRRS) без перевірки —
    4-й контакт може конфліктувати
  - Не залишай скрипт працювати коли підключаєш/відключаєш кабель на ходу —
    щелчки можуть бути інтерпретовані як хибні команди
