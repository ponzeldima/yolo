================================================================================
  ПІДКЛЮЧЕННЯ CRSF ЧЕРЕЗ USB-C — Radiomaster GX12 + macOS
================================================================================

Дата: квітень 2026
Модель пульта: Radiomaster GX12 Dual-Band Gemini-X (EdgeTX)
Метод: CRSF-протокол через USB-C (прямий serial через USB)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ВАЖЛИВО: GX12 — НЕ ТЕ САМЕ ЩО TX12!
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Radiomaster GX12 має ТІЛЬКИ один порт — USB-C.
  На GX12 НЕМАЄ:
    ✗ 3.5mm trainer jack
    ✗ JR bay для зовнішніх модулів (тільки Nano bay)
    ✗ Bluetooth

  Тому для передачі trainer-даних з комп'ютера використовуємо
  USB-C напряму — жодного додаткового обладнання не потрібно!


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1. ЩО ПОТРІБНО
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - USB-C кабель (іде в комплекті з GX12!)
  - Mac з USB-C (або USB-A → USB-C адаптер / хаб)
  - Все. Нічого більше купувати не потрібно.

  Ні USB→UART адаптера, ні 3.5mm breakout, ні дротиків, ні пайки.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  2. ЯК ЦЕ ПРАЦЮЄ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Коли GX12 підключений до Mac через USB-C, він з'являється як
  USB CDC (Communications Device Class) — тобто як serial-порт.

  Ланцюг сигналу:

    Mac [USB-C] ──кабель──► GX12 [USB-C] → EdgeTX Trainer Input

  EdgeTX підтримує прийом trainer-даних (CRSF) через USB serial.
  Ми надсилаємо CRSF RC Channels Packed пакети з Mac напряму в GX12.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  3. ПІДКЛЮЧЕННЯ — ПОКРОКОВО
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ❶ Увімкни GX12

  ❷ Підключи USB-C кабель: Mac ↔ GX12
     На екрані GX12 має з'явитися запит режиму USB.
     Обери "USB Serial (VCP)" або "USB Serial (Debug/CLI)".

     ⚠️  НЕ обирай "USB Storage" — це режим флешки!
     ⚠️  НЕ обирай "USB Joystick" — це для симуляторів (пульт → комп'ютер)!
     Нам потрібен саме Serial — для передачі ДАНИХ З комп'ютера В пульт.

  ❸ Перевір serial-порт на Mac:
     Відкрий термінал:

       ls /dev/cu.usbmodem*

     Має з'явитися щось типу:
       /dev/cu.usbmodem14201
     або
       /dev/cu.usbmodem1101

     Якщо не бачиш — спробуй також:
       ls /dev/cu.usb*

  ❹ Запусти скрипт:
       python rc_test_channels.py --method crsf --port /dev/cu.usbmodemXXXX


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  4. НАЛАШТУВАННЯ EdgeTX НА GX12
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Крок 1: Налаштувати USB-режим
  ─────────────────────────────
    SYS (довге натискання) → Hardware → USB Mode:
    Встанови "USB Serial (VCP)" або "CLI" як дефолтний режим USB
    (щоб не обирати кожного разу при підключенні кабелю).

  Крок 2: Увімкнути Trainer Mode
  ──────────────────────────────
    SYS → Trainer:
    - Trainer Mode: "Master / USB"  або  "Master / Serial"
    - Протокол: "CRSF" (якщо є вибір)

    Примітка: В залежності від версії EdgeTX на GX12,
    пункт може називатися:
    - "Master / USB"  (новіші версії EdgeTX ≥2.10)
    - "Master / Serial" підключений до USB VCP
    Спробуй обидва варіанти якщо один не працює.

  Крок 3: Маппінг каналів
  ────────────────────────
    У розділі Trainer:

    CH1 (Roll/Aileron)   → Source: CH1  → Mode: "Replace"
    CH2 (Pitch/Elevator) → Source: CH2  → Mode: "Replace"
    CH3 (Throttle)       → Source: CH3  → Mode: "Replace"
    CH4 (Yaw/Rudder)     → Source: CH4  → Mode: "Replace"

    Порядок каналів (Mode 2, найпоширеніший):
    CH1 = Roll   (правий стік, ліво-право)
    CH2 = Pitch  (правий стік, вперед-назад)
    CH3 = Throttle (лівий стік, вверх-вниз)
    CH4 = Yaw    (лівий стік, ліво-право)

  Крок 4: Перевірка
  ─────────────────
    - Model → Channel Monitor
    - Запусти скрипт:
        python rc_test_channels.py --method crsf --port /dev/cu.usbmodemXXXX
    - Бари CH1–CH4 мають плавно рухатися (синусоїда)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  5. ПАРАМЕТРИ SERIAL (CRSF через USB)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  USB CDC serial зазвичай ігнорує baud rate (дані передаються на
  швидкості USB). Але pyserial все одно потребує значення:

  Baud rate:   420000 (або будь-яке — USB CDC ігнорує)
  Data bits:   8
  Parity:      None
  Stop bits:   1
  Flow control: None

  Частота оновлення: ~250 Hz (пакет кожні 4 мс)
  Тип пакета: RC Channels Packed (0x16)
  Кількість каналів: 16 (ми використовуємо перші 4)

  6. ЩО МОЖЕ ПІТИ НЕ ТАК — ДІАГНОСТИКА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Проблема: Serial-порт не з'являється (ls /dev/cu.usbmodem* пусто)
  ──────────────────────────────────────────────────────────────────
  - Перевір кабель USB-C (деякі кабелі — тільки для заряджання, без data!)
  - Спробуй інший USB-C порт на Mac
  - На экрані GX12 обери "USB Serial (VCP)", а не "USB Storage"
  - macOS не потребує додаткових драйверів для USB CDC — має працювати з коробки

  Проблема: "Permission denied" при відкритті порту
  ─────────────────────────────────────────────────
  Рішення: sudo chmod 666 /dev/cu.usbmodemXXXX
  На macOS зазвичай працює без sudo.

  Проблема: Канали не рухаються на пульті
  ────────────────────────────────────────
  - Перевір Trainer Mode = "Master / USB" або "Master / Serial"
  - Перевір що маппінг каналів = "Replace" (не "Off"!)
  - Спробуй зменшити baud rate в скрипті (наприклад, 115200)
  - Перевір Channel Monitor: якщо бари на 0% — дані не приходять;
    якщо бари хаотичні — протокол не співпадає (CRSF vs SBUS)

  Проблема: Канали зміщені (центр не 1500)
  ─────────────────────────────────────────
  - Перевір маппінг CRSF values: 992 = центр (1500 мкс)
  - Зроби калібрування в Trainer Settings

  Проблема: GX12 показує "USB Joystick" замість "USB Serial"
  ──────────────────────────────────────────────────────────
  - SYS → Hardware → USB Mode → зміни на "Serial (VCP)" або "CLI"
  - Відключи та підключи кабель заново

  7. БЕЗПЕКА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ✅ USB-C підключення — повністю безпечне.
     USB є стандартний інтерфейс з захистом від перенапруги.
     Нічого спалити неможливо при звичайному підключенні кабелем.

  ⚠️  НЕ РОБИ:
  - Не тримай Trainer Mode "Replace" активним під час реального польоту,
    якщо скрипт не запущений — канали будуть на нулях!
  - Не відключай USB-кабель під час активного Trainer без failsafe —
    пульт може надіслати хибні команди на дрон.
  - Перед першим тестом з реальним дроном — ЗНІМИ ПРОПЕЛЕРИ!

  8. QUICK START — ПОКРОКОВО
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Увімкни GX12
  2. Підключи USB-C кабель (Mac ↔ GX12)
  3. На GX12 обери "USB Serial (VCP)"
  4. SYS → Trainer → Master / USB (або Serial) → CRSF
  5. Trainer CH1–CH4 → Replace
  6. На Mac: ls /dev/cu.usbmodem*  →  знайди порт
  7. Запусти: python rc_test_channels.py --method crsf --port /dev/cu.usbmodemXXXX
  8. Перевір Channel Monitor — канали мають рухатися синусоїдою

  Ніяких адаптерів, дротиків, пайки чи breakout board не потрібно!
