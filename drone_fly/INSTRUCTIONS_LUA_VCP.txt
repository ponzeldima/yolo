  КРОК 1: СКОПІЮВАТИ LUA-СКРИПТ НА SD-КАРТУ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Файл: vcp_bridge.lua  (з цього проекту)
  Куди: SD-карта GX12 → SCRIPTS/MIXES/vcp_bridge.lua

  Способи скопіювати:
    а) Вийняти SD-карту з GX12, підключити до ПК через кард-рідер
    б) Підключити GX12 по USB-C, обрати "USB Storage",
       скопіювати файл, відключити, перезавантажити

  КРОК 2: НАЛАШТУВАТИ USB-VCP РЕЖИМ → LUA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  На GX12:
    SYS (довге натискання) → Hardware → USB-VCP Mode → LUA

  ⚠️  НЕ обирай "Telem Mirror" або "CLI" — тільки "LUA"!

  Це означає: коли ви підключите USB-C кабель і оберете "USB Serial (VCP)",
  серійний порт буде з'єднано з Lua-скриптами через serialRead()/serialWrite().


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  КРОК 3: АКТИВУВАТИ LUA MIXER СКРИПТ У МОДЕЛІ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  На GX12:
    MDL (натиснути) → Custom Scripts (або Global Functions)
    Додати vcp_bridge

  Або для прямого використання в мікшері:
    MDL → Mixes
    Для каналу CH1 (Throttle): замінити джерело на LUA1a (vThr)
    Для каналу CH2 (Roll):     замінити на LUA1b (vRol)
    Для каналу CH3 (Pitch):    замінити на LUA1c (vPit)
    Для каналу CH4 (Yaw):      замінити на LUA1d (vYaw)

  Примітка: LUA1a/b/c/d — це 4 виходи скрипта vcp_bridge.lua
  (визначені як output={"vThr", "vRol", "vPit", "vYaw"})

  КРОК 4: ПІДКЛЮЧИТИ USB-C ТА ЗАПУСТИТИ ТЕСТ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ❶ Увімкни GX12 (без USB)
  ❷ Підключи USB-C кабель
  ❸ На екрані GX12 обери "USB Serial (VCP)"
  ❹ Знайди COM-порт:
     Windows: Device Manager → Ports (COM & LPT) → знайди COM-порт GX12
     macOS:   ls /dev/cu.usbmodem*

  ❺ Запусти тест:

     Windows:
       python rc_test_channels.py --method lua --port COM5

     macOS:
       python rc_test_channels.py --method lua --port /dev/cu.usbmodem14201

  ❻ На пульті GX12 перевір:
     - Зайди в Channel Monitor (MDL → довге натискання або SYS → Channels)
     - Канали мають рухатися синусоїдою!

  ПРОТОКОЛ (для розуміння)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  11 байт на фрейм, 50 Hz (550 байт/с):

  Байт  Зміст
  ────  ──────────────────────────────
  0     0x55 — header byte 1
  1     0xAA — header byte 2
  2-3   Throttle (uint16 big-endian, value + 1024)
  4-5   Roll     (uint16 big-endian, value + 1024)
  6-7   Pitch    (uint16 big-endian, value + 1024)
  8-9   Yaw      (uint16 big-endian, value + 1024)
  10    XOR checksum (байти 2–9)

  Діапазон каналів: -1024..1024 (EdgeTX mixer range)
  Wire encoding: 0..2048 (offset +1024)

  FAILSAFE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Якщо Python-скрипт перестає надсилати дані (crash, disconnect),
  LUA-скрипт автоматично скидає всі канали в 0 (нейтраль) через 200 мс.

  Додатково рекомендується налаштувати Failsafe на самому дроні (FC):
  - Betaflight: Failsafe → Stage 2 → Drop / Land
  - ELRS: Failsafe → No Pulses

  ПОРІВНЯННЯ З CRSF-МЕТОДОМ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  CRSF (rc_crsf_serial.py):
    + Стандартний протокол, працює як trainer input
    + НЕ потребує LUA — працює на БУДЬ-ЯКІЙ прошивці EdgeTX
    + Працює на GX12 (BW 128x64) без Custom Scripts
    - Потрібен режим "Master / USB" + CRSF protocol
    - 420000 baud, складніший протокол (CRC8, 26 байт)

  LUA VCP Bridge (rc_lua_serial.py + vcp_bridge.lua):
    + Простий протокол (11 байт, XOR checksum)
    + Прямий контроль мікшера через LUA
    + Failsafe вбудований в LUA-скрипт
    + Не залежить від trainer mode
    - Потрібен LUA-скрипт на SD-карті
    - Потрібна EdgeTX ≥2.8 для стабільної serialRead()
    - ⚠️ НЕ ПРАЦЮЄ на GX12 з BW-екраном — немає сторінки Custom Scripts!
      Прошивка для BW-пультів (128x64) зазвичай не включає LUA mixer.
      Використовуйте CRSF-метод замість цього.

  АЛЬТЕРНАТИВА: CRSF TRAINER (для GX12 без LUA mixer)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Якщо ваш пульт НЕ має сторінки Custom Scripts (як GX12 BW),
  використовуйте CRSF trainer підхід:

  1. SYS (довге) → PAGE до Trainer:
     - Mode: "Master / USB" (або "Master / Serial")
     - Source/Protocol: CRSF

  2. SYS → Hardware → USB-VCP Mode → CLI або VCP (НЕ LUA)

  3. Підключити USB-C → обрати "USB Serial (VCP)"

  4. Запустити:
     python rc_test_channels.py --method crsf --port COM5

  Дивись: INSTRUCTIONS_CRSF_SERIAL.txt для детальної інструкції.

  TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Канали не рухаються:
    1. Перевір USB-VCP Mode = LUA (SYS → Hardware)
    2. Перевір що скрипт активований в MDL → Custom Scripts
    3. Перевір що Mixes використовують LUA1a/b/c/d як джерело
    4. Перевір COM-порт (Device Manager)

  Канали "смикаються":
    - Перевір кабель USB-C
    - Зменш частоту відправки (збільш sleep)

  "serialRead() returns nil":
    - Переконайся що USB-VCP mode = LUA, а не CLI або Telem Mirror
    - Перезавантаж пульт після зміни режиму

  Python: "Access Denied" на COM-порт:
    - Закрий інші програми що використовують цей порт
    - Спробуй запустити від адміністратора
